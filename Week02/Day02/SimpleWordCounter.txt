// Pseudo-code for Simple Word Counter

// Function to count word frequencies in a given text
function countWordFrequencies(text):
    // 1. Normalize text: convert to lowercase and remove punctuation
    normalizedText = toLowerCase(text)
    normalizedText = replaceAll(normalizedText, "[^a-z\s]", "") // Remove non-alphabetic characters except spaces

    // 2. Split text into words
    words = split(normalizedText, " ") // Split by space

    // 3. Use a Map to store word counts
    wordCounts = new Map<String, Integer>()

    // 4. Iterate through words and count occurrences
    for each word in words:
        if word is not empty:
            if wordCounts.containsKey(word):
                wordCounts.put(word, wordCounts.get(word) + 1)
            else:
                wordCounts.put(word, 1)

    return wordCounts

// Function to find the most frequent word
function findMostFrequentWord(wordCounts):
    mostFrequentWord = ""
    maxCount = 0

    for each entry in wordCounts:
        word = entry.key
        count = entry.value
        if count > maxCount:
            maxCount = count
            mostFrequentWord = word

    return mostFrequentWord + " -> " + maxCount

// Function to find rarely used words (e.g., count of 1)
function findRarelyUsedWords(wordCounts):
    rareWords = new List<String>()
    for each entry in wordCounts:
        word = entry.key
        count = entry.value
        if count == 1:
            rareWords.add(word)
    return rareWords
