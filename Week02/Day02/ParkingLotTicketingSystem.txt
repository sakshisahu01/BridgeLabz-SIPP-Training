// Pseudo-code for Parking Lot Ticketing System

// Superclass: Vehicle
class Vehicle:
    attributes:
        vehicleNumber: string
        entryTime: DateTime

    methods:
        // This method will be overridden by subclasses
        calculateFee(exitTime):
            // Base calculation logic (can be empty or throw error if not overridden)
            return 0

// Subclass: Car
class Car inherits Vehicle:
    constants:
        BASE_FEE: 50
        HOURLY_RATE: 30

    methods:
        calculateFee(exitTime):
            durationHours = ceil( (exitTime - this.entryTime).totalHours )
            if durationHours <= 1:
                return BASE_FEE
            else:
                return BASE_FEE + ( (durationHours - 1) * HOURLY_RATE )

// Subclass: Bike
